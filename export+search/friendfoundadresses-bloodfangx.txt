#include <idc.idc>

static Rebase(addr)
{
    return addr + (get_imagebase() - 0x400000);
}

static InSeg(ea)
{
    return SegStart(ea) != BADADDR;
}

static NameAddr(addr, name)
{
    auto ea = Rebase(addr);
    if (!InSeg(ea)) {
        Message("[skip] 0x%08X not mapped\n", ea);
        return;
    }
    auto other = get_name_ea(BADADDR, name);
    if (other != BADADDR && other != ea) {
        Message("[fix] name '%s' was at 0x%08X (stale from an earlier run); clearing so 0x%08X can take it\n", name, other, ea);
        MakeNameEx(other, "", 0);
    }
    if (MakeNameEx(ea, name, 0x33) == 0) {
        Message("[name] failed on 0x%08X -> %s (collision?)\n", ea, name);
    }
    Message("[ok] 0x%08X %-48s\n", ea, name);
}

static Data(addr, name, kind)
{
    auto ea = Rebase(addr);
    if (!InSeg(ea)) {
        Message("[skip] 0x%08X not mapped\n", ea);
        return;
    }
    if (kind == "byte")   { create_byte(ea); }
    if (kind == "dword")  { create_dword(ea); }
    if (kind == "qword")  { create_qword(ea); }
    if (kind == "float")  { create_float(ea); }
    if (kind == "double") { create_double(ea); }
    if (kind == "ptr")    { create_dword(ea); }
    NameAddr(addr, name);
}
